# Hash-pinned bootstrap for the base-image pip.
#
# pip ships with the python:slim base image and is NOT a runtime dependency of
# the bridge (which only runs `python main.py` out of /opt/venv), but the vuln
# scanner still sees it, so it is upgraded to a patched release at build time.
# 26.1.2 fixed CVE-2026-3219, CVE-2026-6357 and PYSEC-2026-196; 26.2 adds
# CVE-2026-13346 (double-decoded URL path when deriving a link filename) plus
# hardened tar validation against symlink traversal, and 26.2.1 is the keyring
# regression fix on top of it.
#
# It lives in its own file rather than inline in the Dockerfile so the install
# can run under `--require-hashes`: an exact version alone still lets a
# compromised index serve different bytes under that version number. Both the
# wheel and the sdist digest are listed; pip verifies whichever file it picks.
#
# To bump: take the digests from https://pypi.org/pypi/pip/<version>/json
# (`urls[].digests.sha256`) — never from a local download.
pip==26.2.2 \
    --hash=sha256:3a8e5c800492787f33a4114ef54444546c340df428974ebba17e0fd47e11af2f \
    --hash=sha256:fd7f4dfa8190e014013b9f08f3e973f4e4c26ffd179a204895533a7254813a70
